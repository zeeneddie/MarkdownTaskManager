# File-per-Item Architecture

**Datum:** 2025-11-12
**Versie:** 2.0 - Individual Files

---

## 🎯 Design Principe

**In plaats van één grote project.md file, heeft elk Epic, Feature, Story en Task zijn eigen markdown file. Dit maakt het systeem schaalbaar, git-friendly en parallel bewerkbaar.**

---

- ✅ Schaalbaar - 1000+ items geen probleem

**Nadelen:**
- ❌ Geen visuele hierarchie in file system
- ❌ Moet relaties bijhouden in metadata

```python
class MultiFileProjectParser:
    """Parse project from multiple markdown files"""

    def __init__(self, project_root: Path):
        self.project_root = project_root
        self.epics_dir = project_root / "epics"
        self.features_dir = project_root / "features"
        self.stories_dir = project_root / "stories"
        self.tasks_dir = project_root / "tasks"

    def parse_project(self) -> Dict:
        """Parse entire project structure"""

        project = {
            "epics": [],
            "features": [],
            "stories": [],
            "tasks": []
        }

        # Parse all epics
        for epic_file in self.epics_dir.glob("EPIC-*.md"):
            epic = self.parse_epic(epic_file)
            project["epics"].append(epic)

        # Parse all features
        for feature_file in self.features_dir.glob("FEATURE-*.md"):
            feature = self.parse_feature(feature_file)
            project["features"].append(feature)

        # Parse all stories
        for story_file in self.stories_dir.glob("STORY-*.md"):
            story = self.parse_story(story_file)
            project["stories"].append(story)

        # Parse all tasks
        for task_file in self.tasks_dir.glob("TASK-*.md"):
            task = self.parse_task(task_file)
            project["tasks"].append(task)

        # Build relationships
        project = self._build_relationships(project)

        return project

    def parse_epic(self, file_path: Path) -> Dict:
        """Parse single epic markdown file"""

        with open(file_path, 'r') as f:
            content = f.read()

        # Extract YAML frontmatter
        metadata = self._extract_frontmatter(content)

        # Extract markdown content
        body = self._extract_body(content)

        # Parse body sections
        sections = self._parse_sections(body)

        return {
            **metadata,
            "description": sections.get("description"),
            "business_value": sections.get("business_value"),
            "design_notes": sections.get("design_notes"),
            "features": self._extract_links(sections.get("features"), "FEATURE"),
            "file_path": str(file_path)
        }

    def _extract_frontmatter(self, content: str) -> Dict:
        """Extract YAML frontmatter from markdown"""

        import yaml

        # Match: ---\n...\n---
        pattern = r'^---\n(.*?)\n---'
        match = re.search(pattern, content, re.DOTALL)

        if not match:
            return {}

        yaml_content = match.group(1)
        return yaml.safe_load(yaml_content)

    def _extract_links(self, text: str, prefix: str) -> List[str]:
        """Extract item IDs from markdown links"""

        # Match: [FEATURE-001](../features/FEATURE-001.md)
        pattern = rf'\[({prefix}-\d+)\]'
        matches = re.findall(pattern, text or '')

        return matches
```

### Generator: Multi-File Writer

```python
class MultiFileProjectGenerator:
    """Generate markdown files for project items"""

    def __init__(self, project_root: Path):
        self.project_root = project_root
        self.epics_dir = project_root / "epics"
        self.features_dir = project_root / "features"
        self.stories_dir = project_root / "stories"
        self.tasks_dir = project_root / "tasks"

        # Create directories
        self.epics_dir.mkdir(exist_ok=True)
        self.features_dir.mkdir(exist_ok=True)
        self.stories_dir.mkdir(exist_ok=True)
        self.tasks_dir.mkdir(exist_ok=True)

    def generate_epic(self, epic: Dict):
        """Generate epic markdown file"""

        file_path = self.epics_dir / f"{epic['id']}.md"

        # Generate frontmatter
        frontmatter = self._generate_frontmatter(epic, [
            'id', 'type', 'title', 'status', 'priority', 'phase',
            'owner', 'created_at', 'updated_at', 'started_at',
            'target_date', 'sp_total', 'sp_completed', 'progress',
            'epic_type', 'tags', 'dependencies'
        ])

        # Generate body
        body = self._generate_epic_body(epic)

        # Write file
        content = f"---\n{frontmatter}\n---\n\n{body}"

        with open(file_path, 'w') as f:
            f.write(content)

    def _generate_frontmatter(self, data: Dict, fields: List[str]) -> str:
        """Generate YAML frontmatter"""

        import yaml

        frontmatter_data = {}
        for field in fields:
            if field in data:
                value = data[field]

                # Convert datetime to string
                if isinstance(value, datetime):
                    value = value.strftime('%Y-%m-%d')

                frontmatter_data[field] = value

        return yaml.dump(frontmatter_data, default_flow_style=False)

    def _generate_epic_body(self, epic: Dict) -> str:
        """Generate epic markdown body"""

        body = f"# {epic['id']}: {epic['title']}\n\n"

        # Summary
        body += "## 📊 Summary\n\n"
        body += f"{epic.get('description', '')}\n\n"

        # Business Value
        if epic.get('business_value'):
            body += "## 🎯 Business Value\n\n"
            body += f"{epic['business_value']}\n\n"

        # Features
        body += "## 📋 Features\n\n"
        for feature in epic.get('features', []):
            feature_data = self._get_feature(feature['id'])
            status_icon = self._get_status_icon(feature_data['status'])
            body += f"- [{feature['id']}](../features/{feature['id']}.md) - "
            body += f"{feature_data['title']} ({feature_data['sp_total']} FP, {status_icon} {feature_data['status']})\n"

        # Metrics
        body += f"\n## 📈 Metrics\n\n"
        body += f"- **Story Points:** {epic['sp_total']} FP\n"
        body += f"- **T-shirt Size:** {self._calculate_tshirt_size(epic['sp_total'])}\n"
        body += f"- **Estimated Sprints:** {epic.get('estimated_sprints', 'TBD')}\n"
        body += f"- **Confidence:** ±15%\n\n"

        # Footer
        body += "---\n\n"
        body += f"<!-- Auto-generated by Sync Engine -->\n"
        body += f"**Last Sync:** {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}\n"

        return body
```

```python
class MultiFileWatcher:
    """Watch multiple directories for changes"""

    def __init__(self, sync_engine, project_root: Path):
        self.sync_engine = sync_engine
        self.project_root = project_root
        self.observer = Observer()

    def start(self):
        """Start watching all item directories"""

        directories = [
            self.project_root / "epics",
            self.project_root / "features",
            self.project_root / "stories",
            self.project_root / "tasks"
        ]

        handler = ProjectFileHandler(self.sync_engine)

        for directory in directories:
            if directory.exists():
                self.observer.schedule(handler, str(directory), recursive=False)
                print(f"👀 Watching: {directory}")

        self.observer.start()

class ProjectFileHandler(FileSystemEventHandler):
    """Handle file changes"""

    def __init__(self, sync_engine):
        self.sync_engine = sync_engine
        self.last_modified = {}

    def on_modified(self, event):
        if event.is_directory:
            return

        file_path = Path(event.src_path)

        # Only process markdown files
        if file_path.suffix != '.md':
            return

        # Debounce
        current_time = time.time()
        last_mod = self.last_modified.get(str(file_path), 0)

        if current_time - last_mod < 2:  # 2 second debounce
            return

        self.last_modified[str(file_path)] = current_time

        # Determine item type from file path
        if "epics" in file_path.parts:
            item_type = "epic"
        elif "features" in file_path.parts:
            item_type = "feature"
        elif "stories" in file_path.parts:
            item_type = "story"
        elif "tasks" in file_path.parts:
            item_type = "task"
        else:
            return

        print(f"📝 {item_type.upper()} changed: {file_path.name}")

        # Sync to database
        self.sync_engine.sync_file(file_path, item_type)
```

## 📊 Voordelen van File-per-Item

### 1. Git-Friendly

**Before (één file):**
```bash
git diff project.md

# 200 lines changed (mixed epics, features, stories)
# Moeilijk te reviewen
```

**After (per-item files):**
```bash
git diff stories/STORY-001.md

# 5 lines changed (alleen story wijzigingen)
# Makkelijk te reviewen
```

### 2. Parallel Editing

**Before:**
```
Dev 1: Edit STORY-001 in project.md
Dev 2: Edit STORY-002 in project.md
→ Merge conflict in project.md
```

**After:**
```
Dev 1: Edit stories/STORY-001.md
Dev 2: Edit stories/STORY-002.md
→ No conflict! ✅
```

### 3. Performance

**Before:**
```
Parse project.md (10 MB, 500 epics)
→ 5 seconds to parse
```

**After:**
```
Parse stories/STORY-001.md (5 KB)
→ 0.01 seconds to parse
```

### 4. Agents Efficiency

**Before:**
```
Agent needs to read entire project.md to find STORY-001
→ Parse 10 MB
```

**After:**
```
Agent reads stories/STORY-001.md directly
→ Parse 5 KB (2000x faster!)
```
